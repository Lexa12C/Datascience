-- =====================================================================
-- Normalización FIFA World Cup
-- Lee datos desde el esquema plano `fifaworldcup` (ya cargado con
-- 1_cargarcsv.py) y construye el esquema normalizado
-- `fifaworldcup_normalized` con las 15 tablas pedidas.
-- Se usa un esquema nuevo porque el origen ya tiene una tabla llamada
-- `matches`; así no se pisa el dato crudo que se necesita para poblar.
-- =====================================================================

DROP DATABASE IF EXISTS fifaworldcup_normalized;
CREATE DATABASE fifaworldcup_normalized CHARACTER SET utf8mb4;
USE fifaworldcup_normalized;

-- ---------------------------------------------------------------------
-- confederation
-- ---------------------------------------------------------------------
CREATE TABLE confederation (
  confederation_id VARCHAR(10) NOT NULL,
  confederation_name VARCHAR(150) NOT NULL,
  confederation_code VARCHAR(10),
  confederation_wikipedia_link VARCHAR(255),
  PRIMARY KEY (confederation_id)
) ENGINE=InnoDB;

INSERT INTO confederation (confederation_id, confederation_name, confederation_code, confederation_wikipedia_link)
SELECT confederation_id, MAX(confederation_name), MAX(confederation_code), MAX(confederation_wikipedia_link)
FROM fifaworldcup.confederations
WHERE confederation_id IS NOT NULL
GROUP BY confederation_id;

-- ---------------------------------------------------------------------
-- region
-- ---------------------------------------------------------------------
CREATE TABLE region (
  region_id INT NOT NULL AUTO_INCREMENT,
  region_name VARCHAR(100) NOT NULL,
  confederation_id VARCHAR(10),
  PRIMARY KEY (region_id),
  UNIQUE KEY uq_region (region_name, confederation_id),
  CONSTRAINT fk_region_confederation FOREIGN KEY (confederation_id)
    REFERENCES confederation (confederation_id)
) ENGINE=InnoDB;

INSERT INTO region (region_name, confederation_id)
SELECT DISTINCT t.region_name, t.confederation_id
FROM fifaworldcup.teams t
WHERE t.region_name IS NOT NULL AND t.region_name <> '';

-- ---------------------------------------------------------------------
-- country
-- ---------------------------------------------------------------------
CREATE TABLE country (
  country_id INT NOT NULL AUTO_INCREMENT,
  country_name VARCHAR(150) NOT NULL,
  region_id INT,
  PRIMARY KEY (country_id),
  UNIQUE KEY uq_country_name (country_name),
  CONSTRAINT fk_country_region FOREIGN KEY (region_id)
    REFERENCES region (region_id)
) ENGINE=InnoDB;

INSERT INTO country (country_name)
SELECT DISTINCT country_name FROM (
  SELECT country_name FROM fifaworldcup.stadiums
  UNION
  SELECT country_name FROM fifaworldcup.matches
  UNION
  SELECT host_country AS country_name FROM fifaworldcup.tournaments
  UNION
  SELECT team_name AS country_name FROM fifaworldcup.teams
) src
WHERE country_name IS NOT NULL AND country_name <> '';

UPDATE country c
JOIN fifaworldcup.teams t ON t.team_name = c.country_name
JOIN region r ON r.region_name = t.region_name AND r.confederation_id = t.confederation_id
SET c.region_id = r.region_id;

-- ---------------------------------------------------------------------
-- federation
-- ---------------------------------------------------------------------
CREATE TABLE federation (
  federation_id INT NOT NULL AUTO_INCREMENT,
  federation_name VARCHAR(150) NOT NULL,
  federation_wikipedia_link VARCHAR(255),
  confederation_id VARCHAR(10),
  PRIMARY KEY (federation_id),
  UNIQUE KEY uq_federation_name (federation_name),
  CONSTRAINT fk_federation_confederation FOREIGN KEY (confederation_id)
    REFERENCES confederation (confederation_id)
) ENGINE=InnoDB;

INSERT INTO federation (federation_name, federation_wikipedia_link, confederation_id)
SELECT federation_name, MAX(federation_wikipedia_link), MAX(confederation_id)
FROM fifaworldcup.teams
WHERE federation_name IS NOT NULL AND federation_name <> ''
GROUP BY federation_name;

-- ---------------------------------------------------------------------
-- team
-- ---------------------------------------------------------------------
CREATE TABLE team (
  team_id VARCHAR(10) NOT NULL,
  team_name VARCHAR(100) NOT NULL,
  team_code VARCHAR(10),
  mens_team TINYINT(1) DEFAULT 0,
  womens_team TINYINT(1) DEFAULT 0,
  federation_id INT,
  country_id INT,
  mens_team_wikipedia_link VARCHAR(255),
  womens_team_wikipedia_link VARCHAR(255),
  PRIMARY KEY (team_id),
  CONSTRAINT fk_team_federation FOREIGN KEY (federation_id)
    REFERENCES federation (federation_id),
  CONSTRAINT fk_team_country FOREIGN KEY (country_id)
    REFERENCES country (country_id)
) ENGINE=InnoDB;

INSERT INTO team (team_id, team_name, team_code, mens_team, womens_team, federation_id, country_id,
                   mens_team_wikipedia_link, womens_team_wikipedia_link)
SELECT t.team_id, t.team_name, t.team_code, t.mens_team, t.womens_team,
       f.federation_id, c.country_id,
       t.mens_team_wikipedia_link, t.womens_team_wikipedia_link
FROM fifaworldcup.teams t
LEFT JOIN federation f ON f.federation_name = t.federation_name
LEFT JOIN country c ON c.country_name = t.team_name;

-- ---------------------------------------------------------------------
-- city
-- ---------------------------------------------------------------------
CREATE TABLE city (
  city_id INT NOT NULL AUTO_INCREMENT,
  city_name VARCHAR(150) NOT NULL,
  country_id INT,
  city_wikipedia_link VARCHAR(255),
  PRIMARY KEY (city_id),
  UNIQUE KEY uq_city (city_name, country_id),
  CONSTRAINT fk_city_country FOREIGN KEY (country_id)
    REFERENCES country (country_id)
) ENGINE=InnoDB;

INSERT INTO city (city_name, country_id, city_wikipedia_link)
SELECT s.city_name, c.country_id, MAX(s.city_wikipedia_link)
FROM fifaworldcup.stadiums s
LEFT JOIN country c ON c.country_name = s.country_name
WHERE s.city_name IS NOT NULL AND s.city_name <> ''
GROUP BY s.city_name, c.country_id;

-- ---------------------------------------------------------------------
-- stadium
-- ---------------------------------------------------------------------
CREATE TABLE stadium (
  stadium_id VARCHAR(10) NOT NULL,
  stadium_name VARCHAR(150) NOT NULL,
  city_id INT,
  stadium_capacity INT,
  stadium_wikipedia_link VARCHAR(255),
  PRIMARY KEY (stadium_id),
  CONSTRAINT fk_stadium_city FOREIGN KEY (city_id)
    REFERENCES city (city_id)
) ENGINE=InnoDB;

INSERT INTO stadium (stadium_id, stadium_name, city_id, stadium_capacity, stadium_wikipedia_link)
SELECT s.stadium_id, s.stadium_name, ci.city_id, s.stadium_capacity, s.stadium_wikipedia_link
FROM fifaworldcup.stadiums s
LEFT JOIN country co ON co.country_name = s.country_name
LEFT JOIN city ci ON ci.city_name = s.city_name
  AND (ci.country_id = co.country_id OR (ci.country_id IS NULL AND co.country_id IS NULL));

-- ---------------------------------------------------------------------
-- tournament
-- ---------------------------------------------------------------------
CREATE TABLE tournament (
  tournament_id VARCHAR(10) NOT NULL,
  tournament_name VARCHAR(150) NOT NULL,
  `year` YEAR,
  start_date DATE,
  end_date DATE,
  host_country_id INT,
  winner_team_id VARCHAR(10),
  host_won TINYINT(1),
  count_teams INT,
  group_stage TINYINT(1),
  second_group_stage TINYINT(1),
  final_round TINYINT(1),
  round_of_16 TINYINT(1),
  quarter_finals TINYINT(1),
  semi_finals TINYINT(1),
  third_place_match TINYINT(1),
  `final` TINYINT(1),
  PRIMARY KEY (tournament_id),
  CONSTRAINT fk_tournament_host FOREIGN KEY (host_country_id)
    REFERENCES country (country_id),
  CONSTRAINT fk_tournament_winner FOREIGN KEY (winner_team_id)
    REFERENCES team (team_id)
) ENGINE=InnoDB;

INSERT INTO tournament (tournament_id, tournament_name, `year`, start_date, end_date, host_country_id,
                         winner_team_id, host_won, count_teams, group_stage, second_group_stage,
                         final_round, round_of_16, quarter_finals, semi_finals, third_place_match, `final`)
SELECT tr.tournament_id, tr.tournament_name, tr.`year`, NULLIF(tr.start_date,''), NULLIF(tr.end_date,''),
       co.country_id, tm.team_id,
       tr.host_won, tr.count_teams, tr.group_stage, tr.second_group_stage, tr.final_round,
       tr.round_of_16, tr.quarter_finals, tr.semi_finals, tr.third_place_match, tr.`final`
FROM fifaworldcup.tournaments tr
LEFT JOIN country co ON co.country_name = tr.host_country
LEFT JOIN team tm ON tm.team_name = tr.winner;

-- ---------------------------------------------------------------------
-- matches
-- ---------------------------------------------------------------------
CREATE TABLE matches (
  match_id VARCHAR(20) NOT NULL,
  tournament_id VARCHAR(10),
  match_name VARCHAR(150),
  stage_name VARCHAR(50),
  group_name VARCHAR(20),
  group_stage TINYINT(1),
  knockout_stage TINYINT(1),
  match_date DATE,
  match_time TIME,
  stadium_id VARCHAR(10),
  home_team_id VARCHAR(10),
  away_team_id VARCHAR(10),
  home_team_score INT,
  away_team_score INT,
  extra_time TINYINT(1),
  penalty_shootout TINYINT(1),
  home_team_score_penalties INT,
  away_team_score_penalties INT,
  result VARCHAR(50),
  PRIMARY KEY (match_id),
  CONSTRAINT fk_match_tournament FOREIGN KEY (tournament_id)
    REFERENCES tournament (tournament_id),
  CONSTRAINT fk_match_stadium FOREIGN KEY (stadium_id)
    REFERENCES stadium (stadium_id),
  CONSTRAINT fk_match_home_team FOREIGN KEY (home_team_id)
    REFERENCES team (team_id),
  CONSTRAINT fk_match_away_team FOREIGN KEY (away_team_id)
    REFERENCES team (team_id)
) ENGINE=InnoDB;

INSERT INTO matches (match_id, tournament_id, match_name, stage_name, group_name, group_stage,
                      knockout_stage, match_date, match_time, stadium_id, home_team_id, away_team_id,
                      home_team_score, away_team_score, extra_time, penalty_shootout,
                      home_team_score_penalties, away_team_score_penalties, result)
SELECT match_id, tournament_id, match_name, stage_name, group_name, group_stage,
       knockout_stage, NULLIF(match_date,''), NULLIF(match_time,''), stadium_id, home_team_id, away_team_id,
       home_team_score, away_team_score, extra_time, penalty_shootout,
       home_team_score_penalties, away_team_score_penalties, result
FROM fifaworldcup.matches;

-- ---------------------------------------------------------------------
-- position
-- ---------------------------------------------------------------------
CREATE TABLE `position` (
  position_id INT NOT NULL AUTO_INCREMENT,
  position_name VARCHAR(50) NOT NULL,
  position_code VARCHAR(5),
  PRIMARY KEY (position_id),
  UNIQUE KEY uq_position_name (position_name)
) ENGINE=InnoDB;

INSERT INTO `position` (position_name, position_code)
SELECT position_name, MAX(position_code) FROM (
  SELECT position_name, position_code FROM fifaworldcup.player_appearances
  UNION ALL
  SELECT position_name, position_code FROM fifaworldcup.squads
) p
WHERE position_name IS NOT NULL AND position_name <> ''
GROUP BY position_name;

-- ---------------------------------------------------------------------
-- player
-- ---------------------------------------------------------------------
CREATE TABLE player (
  player_id VARCHAR(15) NOT NULL,
  family_name VARCHAR(100),
  given_name VARCHAR(100),
  birth_date DATE,
  female TINYINT(1),
  goal_keeper TINYINT(1),
  defender TINYINT(1),
  midfielder TINYINT(1),
  forward TINYINT(1),
  count_tournaments INT,
  list_tournaments VARCHAR(255),
  player_wikipedia_link VARCHAR(255),
  PRIMARY KEY (player_id)
) ENGINE=InnoDB;

INSERT INTO player (player_id, family_name, given_name, birth_date, female, goal_keeper, defender,
                     midfielder, forward, count_tournaments, list_tournaments, player_wikipedia_link)
SELECT player_id, family_name, given_name, NULLIF(birth_date, ''), female, goal_keeper, defender,
       midfielder, forward, count_tournaments, list_tournaments, player_wikipedia_link
FROM fifaworldcup.players;

-- ---------------------------------------------------------------------
-- player_appearance
-- ---------------------------------------------------------------------
CREATE TABLE player_appearance (
  player_appearance_id INT NOT NULL AUTO_INCREMENT,
  match_id VARCHAR(20) NOT NULL,
  player_id VARCHAR(15) NOT NULL,
  team_id VARCHAR(10),
  position_id INT,
  shirt_number INT,
  starter TINYINT(1),
  substitute TINYINT(1),
  PRIMARY KEY (player_appearance_id),
  UNIQUE KEY uq_match_player (match_id, player_id),
  CONSTRAINT fk_pa_match FOREIGN KEY (match_id)
    REFERENCES matches (match_id),
  CONSTRAINT fk_pa_player FOREIGN KEY (player_id)
    REFERENCES player (player_id),
  CONSTRAINT fk_pa_team FOREIGN KEY (team_id)
    REFERENCES team (team_id),
  CONSTRAINT fk_pa_position FOREIGN KEY (position_id)
    REFERENCES `position` (position_id)
) ENGINE=InnoDB;

INSERT INTO player_appearance (match_id, player_id, team_id, position_id, shirt_number, starter, substitute)
SELECT pa.match_id, pa.player_id, pa.team_id, po.position_id, pa.shirt_number, pa.starter, pa.substitute
FROM fifaworldcup.player_appearances pa
LEFT JOIN `position` po ON po.position_name = pa.position_name;

-- ---------------------------------------------------------------------
-- goal
-- ---------------------------------------------------------------------
CREATE TABLE goal (
  goal_id VARCHAR(10) NOT NULL,
  match_id VARCHAR(20),
  player_id VARCHAR(15),
  team_id VARCHAR(10),
  minute_regulation INT,
  minute_stoppage INT,
  match_period VARCHAR(50),
  own_goal TINYINT(1),
  penalty TINYINT(1),
  PRIMARY KEY (goal_id),
  CONSTRAINT fk_goal_match FOREIGN KEY (match_id)
    REFERENCES matches (match_id),
  CONSTRAINT fk_goal_player FOREIGN KEY (player_id)
    REFERENCES player (player_id),
  CONSTRAINT fk_goal_team FOREIGN KEY (team_id)
    REFERENCES team (team_id)
) ENGINE=InnoDB;

INSERT INTO goal (goal_id, match_id, player_id, team_id, minute_regulation, minute_stoppage,
                   match_period, own_goal, penalty)
SELECT goal_id, match_id, player_id, player_team_id, minute_regulation, minute_stoppage,
       match_period, own_goal, penalty
FROM fifaworldcup.goals;

-- ---------------------------------------------------------------------
-- award
-- ---------------------------------------------------------------------
CREATE TABLE award (
  award_id VARCHAR(10) NOT NULL,
  award_name VARCHAR(100) NOT NULL,
  award_description VARCHAR(255),
  year_introduced YEAR,
  PRIMARY KEY (award_id)
) ENGINE=InnoDB;

INSERT INTO award (award_id, award_name, award_description, year_introduced)
SELECT award_id, award_name, award_description, year_introduced
FROM fifaworldcup.awards;

-- ---------------------------------------------------------------------
-- award_winner
-- ---------------------------------------------------------------------
CREATE TABLE award_winner (
  award_winner_id INT NOT NULL AUTO_INCREMENT,
  tournament_id VARCHAR(10),
  award_id VARCHAR(10),
  shared TINYINT(1),
  player_id VARCHAR(15),
  team_id VARCHAR(10),
  PRIMARY KEY (award_winner_id),
  CONSTRAINT fk_aw_tournament FOREIGN KEY (tournament_id)
    REFERENCES tournament (tournament_id),
  CONSTRAINT fk_aw_award FOREIGN KEY (award_id)
    REFERENCES award (award_id),
  CONSTRAINT fk_aw_player FOREIGN KEY (player_id)
    REFERENCES player (player_id),
  CONSTRAINT fk_aw_team FOREIGN KEY (team_id)
    REFERENCES team (team_id)
) ENGINE=InnoDB;

INSERT INTO award_winner (tournament_id, award_id, shared, player_id, team_id)
SELECT tournament_id, award_id, shared, player_id, team_id
FROM fifaworldcup.award_winners;
